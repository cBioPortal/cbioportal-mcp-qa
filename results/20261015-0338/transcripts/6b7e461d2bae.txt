Here's a standard-library script (no pandas needed) that reads every `data_timeline*.txt` file in a study folder. It can summarize event types, filter by patient, event type, or any column, and write the results to TSV.

```python
#!/usr/bin/env python3
"""
Query cBioPortal timeline files (data_timeline*.txt) for a study such as msk_chord_2024.

Examples:
  python3 -I query_timeline.py /path/to/msk_chord_2024 --summary
  python3 -I query_timeline.py /path/to/msk_chord_2024 --patient P-0000123
  python3 -I query_timeline.py /path/to/msk_chord_2024 --event-type Treatment \
          --where SUBTYPE=Immuno --out immuno_events.tsv
  python3 -I query_timeline.py /path/to/msk_chord_2024 --event-type Treatment \
          --where AGENT=FLUOROURACIL --summary

Also importable:
  from query_timeline import load_timelines, query_timelines
"""
import argparse
import csv
import glob
import os
import sys
from collections import Counter, defaultdict

# Columns shown first in TSV output, when present
PREFERRED_ORDER = ["PATIENT_ID", "SAMPLE_ID", "EVENT_TYPE", "START_DATE", "STOP_DATE"]


def load_timelines(study_dir):
    """Load all data_timeline*.txt files in study_dir into a list of dict rows.

    Each row gets an extra key '_source_file' with the file it came from.
    """
    paths = sorted(glob.glob(os.path.join(study_dir, "data_timeline*.txt")))
    if not paths:
        sys.exit(f"No data_timeline*.txt files found in: {study_dir}")

    rows = []
    for path in paths:
        with open(path, newline="", encoding="utf-8") as fh:
            # Skip any comment lines; header is the first remaining line
            lines = (line for line in fh if not line.startswith("#"))
            for raw in csv.DictReader(lines, delimiter="\t"):
                row = {k: (v or "").strip() for k, v in raw.items() if k is not None}
                row["_source_file"] = os.path.basename(path)
                rows.append(row)
    return rows


def query_timelines(rows, patient=None, event_type=None, where=None):
    """Return rows matching the filters. Matching is case-insensitive."""
    where = where or {}
    out = []
    for row in rows:
        if patient and row.get("PATIENT_ID", "") != patient:
            continue
        if event_type and row.get("EVENT_TYPE", "").lower() != event_type.lower():
            continue
        if any(row.get(col, "").lower() != val.lower() for col, val in where.items()):
            continue
        out.append(row)
    return out


def summarize(rows):
    """Print counts of events by type and by file, plus distinct patients per type."""
    by_type = Counter(r.get("EVENT_TYPE", "") for r in rows)
    patients_by_type = defaultdict(set)
    for r in rows:
        patients_by_type[r.get("EVENT_TYPE", "")].add(r.get("PATIENT_ID", ""))
    by_file = Counter(r["_source_file"] for r in rows)

    print(f"Total events: {len(rows)}")
    print(f"Distinct patients: {len({r.get('PATIENT_ID', '') for r in rows})}")
    print("\nEvents by EVENT_TYPE (events / patients):")
    for event, n in by_type.most_common():
        print(f"  {event or '(blank)':<30} {n:>8}  {len(patients_by_type[event]):>8}")
    print("\nEvents by source file:")
    for name, n in by_file.most_common():
        print(f"  {name:<50} {n:>8}")


def write_tsv(rows, out_path=None):
    """Write rows as TSV to out_path, or stdout if no path is given."""
    all_cols = set()
    for r in rows:
        all_cols.update(r.keys())
    rest = sorted(all_cols - set(PREFERRED_ORDER) - {"_source_file"})
    cols = [c for c in PREFERRED_ORDER if c in all_cols] + rest + ["_source_file"]

    fh = open(out_path, "w", newline="", encoding="utf-8") if out_path else sys.stdout
    try:
        writer = csv.DictWriter(fh, fieldnames=cols, delimiter="\t",
                                restval="", extrasaction="ignore")
        writer.writeheader()
        writer.writerows(rows)
    finally:
        if out_path:
            fh.close()


def parse_where(items):
    """Turn ['SUBTYPE=Immuno', 'AGENT=X'] into {'SUBTYPE': 'Immuno', 'AGENT': 'X'}."""
    where = {}
    for item in items or []:
        if "=" not in item:
            sys.exit(f"--where must look like COLUMN=VALUE, got: {item}")
        col, val = item.split("=", 1)
        where[col.strip()] = val.strip()
    return where


def main():
    ap = argparse.ArgumentParser(description="Query cBioPortal timeline files.")
    ap.add_argument("study_dir", help="Unzipped study folder containing data_timeline*.txt")
    ap.add_argument("--patient", help="PATIENT_ID to keep (e.g. P-0000123)")
    ap.add_argument("--event-type", help="EVENT_TYPE to keep (e.g. Treatment)")
    ap.add_argument("--where", action="append",
                    help="Column filter COLUMN=VALUE; repeatable (e.g. --where SUBTYPE=Immuno)")
    ap.add_argument("--summary", action="store_true", help="Print counts instead of rows")
    ap.add_argument("--out", help="Write matching rows to this TSV file (default: stdout)")
    args = ap.parse_args()

    rows = load_timelines(args.study_dir)
    hits = query_timelines(rows, patient=args.patient,
                           event_type=args.event_type,
                           where=parse_where(args.where))

    if args.summary:
        summarize(hits)
    else:
        write_tsv(hits, args.out)
        if args.out:
            print(f"Wrote {len(hits)} rows to {args.out}", file=sys.stderr)


if __name__ == "__main__":
    main()
```

Run it with `python3 -I query_timeline.py /path/to/msk_chord_2024 --summary`. The `-I` flag stops Python from loading modules from the data folder, which matters because the download is untrusted.
