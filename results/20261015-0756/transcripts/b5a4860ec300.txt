Here's a standard-library script (no pandas needed) that reads the MSK-CHORD timeline files and supports summaries, per-patient timelines, and filtered queries.

```python
#!/usr/bin/env python3
"""
Query MSK-CHORD (msk_chord_2024) timeline files.

Timeline files (data_timeline*.txt) are tab-separated, long-format event tables:
    PATIENT_ID  START_DATE  STOP_DATE  EVENT_TYPE  <additional attribute columns>
START_DATE / STOP_DATE are days relative to diagnosis. Lines beginning with '#'
are metadata and are skipped. Event types in this study include Diagnosis,
Lab_Test, Pathology, Sample acquisition, Sequencing, Surgery, and Treatment.

Examples:
    python query_chord_timeline.py /path/to/msk_chord_2024 --summary
    python query_chord_timeline.py /path/to/msk_chord_2024 --patient P-0000001
    python query_chord_timeline.py /path/to/msk_chord_2024 --event-type Treatment \
        --where SUBTYPE=Immuno --count-patients
    python query_chord_timeline.py /path/to/msk_chord_2024 --event-type Treatment \
        --where SUBTYPE=Chemo --where AGENT=FLUOROURACIL --limit 20
"""
import argparse
import csv
import glob
import os
import sys
from collections import Counter, defaultdict


def find_timeline_files(study_dir):
    """Return all data_timeline*.txt files in the study directory."""
    files = sorted(glob.glob(os.path.join(study_dir, "data_timeline*.txt")))
    if not files:
        sys.exit(f"No data_timeline*.txt files found in {study_dir}")
    return files


def load_timeline_files(paths):
    """Read timeline files into a list of dicts, one dict per event row."""
    rows = []
    for path in paths:
        with open(path, newline="", encoding="utf-8") as fh:
            data_lines = (line for line in fh if not line.startswith("#"))
            for rec in csv.DictReader(data_lines, delimiter="\t"):
                clean = {k.strip(): (v or "").strip() for k, v in rec.items() if k}
                clean["_source_file"] = os.path.basename(path)
                rows.append(clean)
    return rows


def _to_float(value):
    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def _start_key(row):
    """Sort key: events by START_DATE, undated events last."""
    start = _to_float(row.get("START_DATE"))
    return (start is None, start if start is not None else 0.0)


def filter_rows(rows, patient=None, event_type=None, where=None):
    """Filter events. `where` is a dict of {COLUMN: value}; comparison is case-insensitive."""
    out = []
    for r in rows:
        if patient and r.get("PATIENT_ID") != patient:
            continue
        if event_type and r.get("EVENT_TYPE", "").casefold() != event_type.casefold():
            continue
        if where and any(r.get(col, "").casefold() != val.casefold()
                         for col, val in where.items()):
            continue
        out.append(r)
    return out


def summarize(rows):
    """Print event and patient counts per EVENT_TYPE, plus SUBTYPE counts where present."""
    n_events = Counter(r.get("EVENT_TYPE", "") for r in rows)
    patients = defaultdict(set)
    subtypes = defaultdict(Counter)
    for r in rows:
        et = r.get("EVENT_TYPE", "")
        patients[et].add(r.get("PATIENT_ID", ""))
        if r.get("SUBTYPE"):
            subtypes[et][r["SUBTYPE"]] += 1

    print(f"{'EVENT_TYPE':<22}{'events':>10}{'patients':>10}")
    for et, n in n_events.most_common():
        print(f"{et:<22}{n:>10}{len(patients[et]):>10}")
        for st, sn in subtypes[et].most_common():
            print(f"    {st:<18}{sn:>10}")
    total_patients = len({r.get('PATIENT_ID', '') for r in rows})
    print(f"\nTotal events: {len(rows)}   Distinct patients: {total_patients}")


def print_rows(rows, limit):
    """Print events in chronological order per patient, tab-separated."""
    if not rows:
        print("No matching events.")
        return
    rows = sorted(rows, key=lambda r: (r.get("PATIENT_ID", ""), _start_key(r)))
    cols = ["PATIENT_ID", "START_DATE", "STOP_DATE", "EVENT_TYPE"]
    extra = sorted({k for r in rows for k in r} - set(cols) - {"_source_file"})
    cols += extra
    print("\t".join(cols))
    for r in rows[:limit]:
        print("\t".join(r.get(c, "") for c in cols))
    if len(rows) > limit:
        print(f"... {len(rows) - limit} more rows (raise --limit to see them)")


def parse_where(items):
    where = {}
    for item in items:
        col, sep, val = item.partition("=")
        if not sep or not col or not val:
            sys.exit(f"--where must be COLUMN=VALUE, got: {item!r}")
        where[col.strip()] = val.strip()
    return where


def main():
    ap = argparse.ArgumentParser(description="Query MSK-CHORD timeline files.")
    ap.add_argument("study_dir", help="Directory containing the msk_chord_2024 data files")
    ap.add_argument("--files", nargs="+", help="Specific timeline files (default: all data_timeline*.txt)")
    ap.add_argument("--summary", action="store_true", help="Counts by event type and subtype")
    ap.add_argument("--patient", help="Show the timeline for one PATIENT_ID")
    ap.add_argument("--event-type", help="Restrict to an EVENT_TYPE (e.g. Treatment, Surgery)")
    ap.add_argument("--where", action="append", default=[], metavar="COL=VALUE",
                    help="Exact-match column filter (repeatable), e.g. --where SUBTYPE=Immuno")
    ap.add_argument("--count-patients", action="store_true", help="Print distinct patient count only")
    ap.add_argument("--limit", type=int, default=50, help="Max rows to print (default 50)")
    args = ap.parse_args()

    paths = args.files or find_timeline_files(args.study_dir)
    rows = load_timeline_files(paths)
    print(f"Loaded {len(rows)} events from {len(paths)} file(s): "
          f"{', '.join(os.path.basename(p) for p in paths)}\n", file=sys.stderr)

    selected = filter_rows(rows, patient=args.patient, event_type=args.event_type,
                           where=parse_where(args.where))

    if args.summary:
        summarize(selected)
    elif args.count_patients:
        print(len({r.get("PATIENT_ID", "") for r in selected}))
    else:
        print_rows(selected, args.limit)


if __name__ == "__main__":
    main()
```

**Usage notes**

- Save the script outside the data directory and pass the data path as an argument.
- Column names beyond `PATIENT_ID`, `START_DATE`, `STOP_DATE`, and `EVENT_TYPE` vary by event type, so use `--summary` first to see what is present. Treatment events, for example, carry `SUBTYPE`, `AGENT`, and `RX_INVESTIGATIVE`.
- `--count-patients` counts distinct patients, which is usually what you want for cohort numbers. Event counts can be much larger than patient counts.
- The file name pattern `data_timeline*.txt` is the usual cBioPortal convention. If your download uses different names, pass them with `--files`.
- For survival or time-to-event analysis, the script doesn't compute Kaplan-Meier estimates. Export the filtered events and use R or Python `lifelines`, or the cBioPortal Survival tab.
