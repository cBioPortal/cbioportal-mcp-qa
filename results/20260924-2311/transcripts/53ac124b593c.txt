"""
MSK-CHORD Clinical Timeline Query Module

This module provides functions to query clinical timeline/event data from the 
MSK-CHORD study using the cBioPortal REST API.
"""

import requests
import pandas as pd
import json
from typing import List, Dict, Optional
from datetime import datetime

# cBioPortal public REST API endpoint
CBIOPORTAL_API = "https://www.cbioportal.org/api"
STUDY_ID = "msk_chord_2024"


class MSKChordTimelineQuery:
    """Query class for MSK-CHORD clinical timeline data"""
    
    def __init__(self, api_base: str = CBIOPORTAL_API):
        self.api_base = api_base
        self.study_id = STUDY_ID
        self.session = requests.Session()
    
    def get_patient_timeline(
        self, 
        patient_id: str,
        include_treatments: bool = True,
        include_diagnoses: bool = True
    ) -> pd.DataFrame:
        """
        Retrieve timeline events for a specific patient.
        
        Args:
            patient_id: Patient identifier (e.g., "P-0000001")
            include_treatments: Include treatment events
            include_diagnoses: Include diagnosis/progression events
        
        Returns:
            DataFrame with timeline events
        """
        endpoint = (
            f"{self.api_base}/patients/{patient_id}/clinical-events"
            f"?studyId={self.study_id}"
        )
        
        try:
            response = self.session.get(endpoint)
            response.raise_for_status()
            events = response.json()
            
            if not events:
                return pd.DataFrame()
            
            # Convert to DataFrame
            df = pd.json_normalize(events)
            
            # Filter by event type if requested
            if include_treatments and include_diagnoses:
                pass  # Keep all
            elif include_treatments:
                df = df[df['eventType'].isin(['TREATMENT', 'Treatment'])]
            elif include_diagnoses:
                df = df[df['eventType'].isin(['DIAGNOSIS', 'Diagnosis', 'PROGRESSION'])]
            
            return df
        
        except requests.exceptions.RequestException as e:
            print(f"Error retrieving timeline for {patient_id}: {e}")
            return pd.DataFrame()
    
    def get_cohort_timeline_summary(
        self,
        event_type: str = "TREATMENT"
    ) -> pd.DataFrame:
        """
        Get a summary of all timeline events of a specific type in the cohort.
        
        Args:
            event_type: Type of event to retrieve (e.g., "TREATMENT", "DIAGNOSIS")
        
        Returns:
            DataFrame with event summary statistics
        """
        endpoint = (
            f"{self.api_base}/studies/{self.study_id}/clinical-event-types"
        )
        
        try:
            response = self.session.get(endpoint)
            response.raise_for_status()
            event_types = response.json()
            
            return pd.DataFrame(event_types)
        
        except requests.exceptions.RequestException as e:
            print(f"Error retrieving event types: {e}")
            return pd.DataFrame()
    
    def get_patient_samples(self, patient_id: str) -> pd.DataFrame:
        """
        Get all samples for a patient.
        
        Args:
            patient_id: Patient identifier
        
        Returns:
            DataFrame with sample information
        """
        endpoint = (
            f"{self.api_base}/patients/{patient_id}/samples"
            f"?studyId={self.study_id}"
        )
        
        try:
            response = self.session.get(endpoint)
            response.raise_for_status()
            samples = response.json()
            
            return pd.DataFrame(samples)
        
        except requests.exceptions.RequestException as e:
            print(f"Error retrieving samples for {patient_id}: {e}")
            return pd.DataFrame()


class DirectClickHouseQuery:
    """
    Query MSK-CHORD timeline data directly via ClickHouse SQL.
    
    NOTE: ClickHouse access requires credentials. Use this class if you have
    direct database access; otherwise use MSKChordTimelineQuery for REST API access.
    """
    
    def __init__(self, host: str, user: str, password: str, database: str = "default"):
        """
        Initialize ClickHouse connection.
        
        Args:
            host: ClickHouse server host
            user: Database username
            password: Database password
            database: Database name (default: "default")
        """
        try:
            from clickhouse_driver import Client
            self.client = Client(host, user=user, password=password, database=database)
        except ImportError:
            raise ImportError("clickhouse-driver required: pip install clickhouse-driver")
    
    def get_patient_timeline_events(self, patient_stable_id: str) -> pd.DataFrame:
        """
        Get all timeline events for a patient (ClickHouse direct query).
        
        Args:
            patient_stable_id: Patient stable ID (e.g., "P-0000001")
        
        Returns:
            DataFrame with clinical events
        """
        query = """
        SELECT 
            patient_stable_id,
            event_type,
            start_date,
            stop_date,
            clinical_event_id
        FROM clinical_event_derived
        WHERE cancer_study_identifier = 'msk_chord_2024'
          AND patient_stable_id = %s
        ORDER BY start_date ASC
        """
        
        rows = self.client.execute(query, parameters=[patient_stable_id])
        df = pd.DataFrame(rows, columns=['patient_id', 'event_type', 'start_date', 'stop_date', 'event_id'])
        return df
    
    def get_treatment_events_for_patient(self, patient_stable_id: str) -> pd.DataFrame:
        """
        Get treatment events with drug details for a patient.
        
        Args:
            patient_stable_id: Patient stable ID
        
        Returns:
            DataFrame with treatment events and agent details
        """
        query = """
        SELECT 
            ce.patient_stable_id,
            ce.start_date,
            ce.stop_date,
            ce.clinical_event_id,
            SUBTYPE.value as treatment_subtype,
            AGENT.value as agent,
            RX_INVESTIGATIVE.value as investigational,
            INFERRED_TX_PROB.value as nlp_confidence
        FROM clinical_event_derived ce
        LEFT JOIN clinical_event_data SUBTYPE 
            ON ce.clinical_event_id = SUBTYPE.clinical_event_id 
            AND SUBTYPE.key = 'SUBTYPE'
        LEFT JOIN clinical_event_data AGENT 
            ON ce.clinical_event_id = AGENT.clinical_event_id 
            AND AGENT.key = 'AGENT'
        LEFT JOIN clinical_event_data RX_INVESTIGATIVE 
            ON ce.clinical_event_id = RX_INVESTIGATIVE.clinical_event_id 
            AND RX_INVESTIGATIVE.key = 'RX_INVESTIGATIVE'
        LEFT JOIN clinical_event_data INFERRED_TX_PROB 
            ON ce.clinical_event_id = INFERRED_TX_PROB.clinical_event_id 
            AND INFERRED_TX_PROB.key = 'INFERRED_TX_PROB'
        WHERE ce.cancer_study_identifier = 'msk_chord_2024'
          AND ce.patient_stable_id = %s
          AND ce.event_type = 'Treatment'
        ORDER BY ce.start_date ASC
        """
        
        rows = self.client.execute(query, parameters=[patient_stable_id])
        columns = ['patient_id', 'start_date', 'stop_date', 'event_id', 
                   'treatment_type', 'agent', 'investigational', 'nlp_confidence']
        df = pd.DataFrame(rows, columns=columns)
        return df
    
    def get_progression_events(self, patient_stable_id: str) -> pd.DataFrame:
        """
        Get progression events (diagnosis with progression status).
        
        Args:
            patient_stable_id: Patient stable ID
        
        Returns:
            DataFrame with progression events
        """
        query = """
        SELECT 
            ce.patient_stable_id,
            ce.start_date,
            ce.stop_date,
            SUBTYPE.value as event_subtype,
            PROGRESSION.value as progression_status
        FROM clinical_event_derived ce
        LEFT JOIN clinical_event_data SUBTYPE 
            ON ce.clinical_event_id = SUBTYPE.clinical_event_id 
            AND SUBTYPE.key = 'SUBTYPE'
        LEFT JOIN clinical_event_data PROGRESSION 
            ON ce.clinical_event_id = PROGRESSION.clinical_event_id 
            AND PROGRESSION.key = 'PROGRESSION'
        WHERE ce.cancer_study_identifier = 'msk_chord_2024'
          AND ce.patient_stable_id = %s
          AND ce.event_type = 'Diagnosis'
        ORDER BY ce.start_date ASC
        """
        
        rows = self.client.execute(query, parameters=[patient_stable_id])
        columns = ['patient_id', 'start_date', 'stop_date', 'event_subtype', 'progression']
        df = pd.DataFrame(rows, columns=columns)
        return df
    
    def get_cohort_treatment_summary(self) -> pd.DataFrame:
        """
        Get summary of all treatments by type in the cohort.
        
        Returns:
            DataFrame with treatment counts by type and agent
        """
        query = """
        SELECT 
            SUBTYPE.value as treatment_type,
            AGENT.value as agent,
            COUNT(DISTINCT ce.patient_stable_id) as patient_count
        FROM clinical_event_derived ce
        LEFT JOIN clinical_event_data SUBTYPE 
            ON ce.clinical_event_id = SUBTYPE.clinical_event_id 
            AND SUBTYPE.key = 'SUBTYPE'
        LEFT JOIN clinical_event_data AGENT 
            ON ce.clinical_event_id = AGENT.clinical_event_id 
            AND AGENT.key = 'AGENT'
        WHERE ce.cancer_study_identifier = 'msk_chord_2024'
          AND ce.event_type = 'Treatment'
        GROUP BY treatment_type, agent
        ORDER BY treatment_type, patient_count DESC
        """
        
        rows = self.client.execute(query)
        columns = ['treatment_type', 'agent', 'patient_count']
        df = pd.DataFrame(rows, columns=columns)
        return df
